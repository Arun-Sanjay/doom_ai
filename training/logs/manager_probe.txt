Manager probe: what the manager's two heads output in the promoted checkpoints
==============================================================================

Measured 2026-11-27 about 5:00 IST on the development Mac, CPU, no environment: the networks are run on
the recorded inputs of the provenance probe (training/rl/probe_episodes.npz, 2 episodes x 40 decisions,
so 10 manager steps) with the script printed at the end of this file. Checkpoints are the files pulled
from the training machine at each check-in (runs/, not tracked); `it` is the iteration of the second
phase (runs/unified_full_2), except the last row, which is the first shipped agent (first phase).
The policy file shipped when this was measured (sha256 e2a54cd164bf, commit 16f3ac7) is the export of
iteration 1000; the row marked `shipped` is that file's checkpoint.

(a) entropy of the manager's own distributions in nats, and the mean probability of its top choice.
    A uniform choice has entropy ln 8 = 2.079 (sub-goal) and ln 4096 = 8.318 (target cell).
(b) how far the controller's head probabilities move (mean total-variation distance per head) when the
    manager's choices are replaced by random other ones on the same recorded inputs.

The same drift over whole rollouts (about 98,000 decisions per iteration) is in the columns ent_subgoal,
ent_target, agree_subgoal and agree_target of unified_full_2_train_log.csv in this folder.

"""Part (a) and (b) of data/plateau/manager_probe.py on more checkpoints, read-only."""
import sys, math
import numpy as np, torch
sys.path.insert(0, ".")
from training import paths  # noqa: F401
from doomnet import spec
from training.rl import deps, provenance
torch.set_num_threads(2)
CK = {"it200 (main_0035)": "runs/vast/checkin/main_0035.pt", "it500 (main_0135)": "runs/vast/checkin/main_0135.pt",
      "it725 (main_0220)": "runs/vast/checkin/main_0220.pt", "it1000 (main_0305, shipped)": "runs/vast/checkin/main_0305.pt",
      "it1225 (main_0355)": "runs/vast/checkin/main_0355.pt", "it1525 (main_0445)": "runs/vast/checkin/main_0445.pt",
      "p1 it425 (unified_it425)": "runs/vast/unified/it_b.pt"}
P = provenance.probe()
INP = {k: torch.from_numpy(P[k]) for k in ("rays", "ents", "ent_mask", "scal", "map", "masks")}
E, T = P["scal"].shape[:2]
def net_of(state, cfg):
    net = deps.make_policy("real", **cfg); net.load_state_dict(state); net.eval(); return net
def run(net, subgoals=None, targets=None, sample=False, seed=0):
    torch.manual_seed(seed)
    probs, ent_s, ent_t, top_s, top_t = [], [], [], [], []
    with torch.no_grad():
        state = net.initial_state(E, torch.device("cpu"))
        for t in range(T):
            m = t // spec.MANAGER_PERIOD
            forced = {}
            if subgoals is not None: forced["subgoal"] = torch.as_tensor(subgoals[:, m])
            if targets is not None: forced["target"] = torch.as_tensor(targets[:, m])
            runm = torch.full((E,), t % spec.MANAGER_PERIOD == 0)
            out, state = net(INP["rays"][:, t], INP["ents"][:, t], INP["ent_mask"][:, t], INP["scal"][:, t], INP["map"][:, t],
                             state=state, masks=INP["masks"][:, t], run_manager=runm, forced=forced or None, sample=sample)
            probs.append(torch.cat([torch.softmax(l.float(), 1) for l in out["logits"]], 1))
            if t % spec.MANAGER_PERIOD == 0:
                ls, lt = torch.log_softmax(out["subgoal_logits"].float(), 1), torch.log_softmax(out["target_logits"].float(), 1)
                ent_s.append(-(ls.exp() * ls).sum(1)); ent_t.append(-(lt.exp() * lt).sum(1))
                top_s.append(ls.exp().max(1).values); top_t.append(lt.exp().max(1).values)
    return torch.stack(probs, 1), torch.stack(ent_s, 1), torch.stack(ent_t, 1), torch.stack(top_s, 1), torch.stack(top_t, 1)
SL = np.cumsum((0,) + spec.HEAD_SIZES)
def tv(a, b):
    return [float(0.5 * (a[..., SL[k]:SL[k + 1]] - b[..., SL[k]:SL[k + 1]]).abs().sum(-1).mean()) for k in range(len(spec.HEAD_SIZES))]
rng = np.random.default_rng(0)
NM = math.ceil(T / spec.MANAGER_PERIOD)
base_s = rng.integers(0, spec.NUM_SUBGOALS, (E, NM)); base_t = rng.integers(0, 4096, (E, NM))
alts = [(rng.integers(0, spec.NUM_SUBGOALS, (E, NM)), rng.integers(0, 4096, (E, NM))) for _ in range(8)]
print(f"probe: {E} episodes x {T} steps; max entropy: sub-goal ln 8 = {math.log(8):.3f}, target ln 4096 = {math.log(4096):.3f}")
nets = {}
for name, path in CK.items():
    c = torch.load(path, map_location="cpu", weights_only=False)
    nets[name] = net_of(c["model"], c["policy_config"])
    _, es, et, ts, tt = run(nets[name], sample=False)
    print(f"(a) {name:30s} it {c.get('iteration')} steps {c.get('env_steps')}: sub-goal entropy {es.mean():.3f} (top p {ts.mean():.3f}) | target entropy {et.mean():.3f} (top p {tt.mean():.4f})")
print("(b) mean total-variation distance per head when the manager's forced choices are swapped (8 random alternatives vs one base)")
print(f"    {'':30} {'changed':16} " + " ".join(f"{h:>7}" for h in spec.HEAD_NAMES))
for name, net in nets.items():
    base = run(net, base_s, base_t)[0]
    for what in ("sub-goal+target", "sub-goal only", "target only"):
        acc = []
        for s, t in alts:
            alt = run(net, s if "sub-goal" in what else base_s, t if "target" in what else base_t)[0]
            acc.append(tv(base, alt))
        print(f"    {name:30} {what:16} " + " ".join(f"{v:7.4f}" for v in np.mean(acc, 0)))
